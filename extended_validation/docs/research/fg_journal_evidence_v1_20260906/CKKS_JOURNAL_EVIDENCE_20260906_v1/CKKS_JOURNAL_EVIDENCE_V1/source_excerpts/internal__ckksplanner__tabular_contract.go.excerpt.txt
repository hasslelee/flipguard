667: func addTabularAffine(
668: 	graph *ir.Graph,
669: 	prefix string,
670: 	inputs []ir.NodeID,
671: 	weights []float64,
672: 	bias float64,
673: ) ir.NodeID {
674: 	terms := make([]ir.NodeID, len(inputs))
675: 	for index, input := range inputs {
676: 		termID := ir.NodeID(fmt.Sprintf("%s_term_%d", prefix, index))
677: 		graph.MustAddNode(
678: 			ir.NewMulConst(
679: 				termID,
680: 				string(termID),
681: 				input,
682: 				weights[index],
683: 			),
684: 		)
685: 		terms[index] = termID
686: 	}
687: 
688: 	accumulator := terms[0]
689: 	for index := 1; index < len(terms); index++ {
690: 		sumID := ir.NodeID(fmt.Sprintf("%s_sum_%d", prefix, index))
691: 		graph.MustAddNode(
692: 			ir.NewBinary(
693: 				sumID,
694: 				string(sumID),
695: 				ir.OpAdd,
696: 				accumulator,
697: 				terms[index],
698: 			),
699: 		)
700: 		accumulator = sumID
701: 	}
702: 
703: 	if bias == 0 {
704: 		return accumulator
705: 	}
706: 
707: 	biasID := ir.NodeID(prefix + "_bias")
708: 	outputID := ir.NodeID(prefix + "_affine")
709: 	graph.MustAddNode(ir.NewConst(biasID, string(biasID), bias))
710: 	graph.MustAddNode(
711: 		ir.NewBinary(
712: 			outputID,
713: 			string(outputID),
714: 			ir.OpAdd,
715: 			accumulator,
716: 			biasID,
717: 		),
718: 	)
719: 	return outputID
720: }

1160: // analyzeLattigoRescaleDemand models the current backend rather than equating
1161: // multiplicative depth with modulus levels. In Lattigo v6, multiplication by a
1162: // non-integer scalar adds the current Q-prime scale. RescaleTo can therefore
1163: // consume several primes in one call.
1164: func analyzeLattigoRescaleDemand(
1165: 	graph *ir.Graph,
1166: ) (lattigoScaleDemand, error) {
1167: 	states := make(map[ir.NodeID]lattigoScaleState, graph.Len())
1168: 
1169: 	for _, node := range graph.Nodes() {
1170: 		inputStates := make(
1171: 			[]lattigoScaleState,
1172: 			0,
1173: 			len(node.Inputs),
1174: 		)
1175: 		for _, inputID := range node.Inputs {
1176: 			state, exists := states[inputID]
1177: 			if !exists {
1178: 				return lattigoScaleDemand{}, fmt.Errorf(
1179: 					"scale trace input %s for node %s is missing",
1180: 					inputID,
1181: 					node.ID,
1182: 				)
1183: 			}
1184: 			inputStates = append(inputStates, state)
1185: 		}
1186: 
1187: 		state := lattigoScaleState{}
1188: 		switch node.Op {
1189: 		case ir.OpInput:
1190: 			state = lattigoScaleState{
1191: 				IsCiphertext:  true,
1192: 				ScaleExponent: 1,
1193: 			}
1194: 
1195: 		case ir.OpConst:
1196: 			state = lattigoScaleState{}
1197: 
1198: 		case ir.OpAdd, ir.OpSub:
1199: 			for _, input := range inputStates {
1200: 				if !input.IsCiphertext {
1201: 					continue
1202: 				}
1203: 				state.IsCiphertext = true
1204: 				state.ScaleExponent = maxInt(
1205: 					state.ScaleExponent,
1206: 					input.ScaleExponent,
1207: 				)
1208: 				state.LevelsConsumed = maxInt(
1209: 					state.LevelsConsumed,
1210: 					input.LevelsConsumed,
1211: 				)
1212: 			}
1213: 			if !state.IsCiphertext {
1214: 				return lattigoScaleDemand{}, fmt.Errorf(
1215: 					"scale trace node %s adds only plaintext values",
1216: 					node.ID,
1217: 				)
1218: 			}
1219: 
1220: 		case ir.OpMulConst:
1221: 			input := inputStates[0]
1222: 			if !input.IsCiphertext {
1223: 				return lattigoScaleDemand{}, fmt.Errorf(
1224: 					"scale trace node %s multiplies a plaintext",
1225: 					node.ID,
1226: 				)
1227: 			}
1228: 			state = input
1229: 			state.ScaleExponent++
1230: 
1231: 		case ir.OpMul:
1232: 			left := inputStates[0]
1233: 			right := inputStates[1]
1234: 			if !left.IsCiphertext || !right.IsCiphertext {
1235: 				return lattigoScaleDemand{}, fmt.Errorf(
1236: 					"scale trace node %s requires ciphertext inputs",
1237: 					node.ID,
1238: 				)
1239: 			}
1240: 			state = lattigoScaleState{
1241: 				IsCiphertext: true,
1242: 				ScaleExponent: left.ScaleExponent +
1243: 					right.ScaleExponent,
1244: 				LevelsConsumed: maxInt(
1245: 					left.LevelsConsumed,
1246: 					right.LevelsConsumed,
1247: 				),
1248: 			}
1249: 
1250: 		case ir.OpPow2:
1251: 			input := inputStates[0]
1252: 			productExponent := 2 * input.ScaleExponent
1253: 			state = lattigoScaleState{
1254: 				IsCiphertext:  true,
1255: 				ScaleExponent: 1,
1256: 				LevelsConsumed: input.LevelsConsumed +
1257: 					productExponent - 1,
1258: 			}
1259: 
1260: 		case ir.OpPow3:
1261: 			return lattigoScaleDemand{}, fmt.Errorf(
1262: 				"standalone pow3 has no declared rescale implementation",
1263: 			)
1264: 
1265: 		case ir.OpPoly:
1266: 			input := inputStates[0]
1267: 			degree := len(node.Coeffs) - 1
1268: 			if degree != 3 {
1269: 				return lattigoScaleDemand{}, fmt.Errorf(
1270: 					"scale trace supports cubic polynomial only, got degree %d",
1271: 					degree,
1272: 				)
1273: 			}
1274: 
1275: 			// Current Horner backend:
1276: 			// z2=rescale(z*z), inner=(-.004*z2)+.197,
1277: 			// y=rescale(inner*z)+.5.
1278: 			cubicLevels := 3 * input.ScaleExponent
1279: 			state = lattigoScaleState{
1280: 				IsCiphertext:  true,
1281: 				ScaleExponent: 1,
1282: 				LevelsConsumed: input.LevelsConsumed +
1283: 					cubicLevels,
1284: 			}
1285: 
1286: 		default:
1287: 			return lattigoScaleDemand{}, fmt.Errorf(
1288: 				"unsupported scale trace op %s",
1289: 				node.Op,
1290: 			)
1291: 		}
1292: 
1293: 		states[node.ID] = state
1294: 	}
1295: 
1296: 	output, exists := states[graph.Output]
1297: 	if !exists || !output.IsCiphertext {
1298: 		return lattigoScaleDemand{}, fmt.Errorf(
1299: 			"scale trace output is not a ciphertext",
1300: 		)
1301: 	}
1302: 
1303: 	return lattigoScaleDemand{
1304: 		LevelsConsumed: output.LevelsConsumed,
1305: 		ScaleExponent:  output.ScaleExponent,
1306: 		RequiredQPrimes: output.LevelsConsumed +
1307: 			output.ScaleExponent,
1308: 	}, nil
1309: }
