102: func CertifyMulticlassArgmaxBounds(
103: 	plainLogits []float64,
104: 	errorBounds []float64,
105: ) (MulticlassBoundCertificate, error) {
106: 	if err := validateLogitVector("plaintext", plainLogits); err != nil {
107: 		return MulticlassBoundCertificate{}, err
108: 	}
109: 	if len(errorBounds) != len(plainLogits) {
110: 		return MulticlassBoundCertificate{}, fmt.Errorf(
111: 			"error-bound count %d does not match class count %d",
112: 			len(errorBounds),
113: 			len(plainLogits),
114: 		)
115: 	}
116: 	for index, bound := range errorBounds {
117: 		if !isNonNegativeFinite(bound) {
118: 			return MulticlassBoundCertificate{}, fmt.Errorf(
119: 				"error bound %d must be finite and non-negative",
120: 				index,
121: 			)
122: 		}
123: 	}
124: 	top, runnerUp, gap, tied := topTwoClasses(plainLogits)
125: 	certificate := MulticlassBoundCertificate{
126: 		SchemaVersion:   MulticlassDecisionContractSchemaV2,
127: 		TieBreak:        DeterministicLowestIndexTieBreak,
128: 		PlainClass:      top,
129: 		RunnerUp:        runnerUp,
130: 		TopTwoGap:       gap,
131: 		PlainTie:        tied,
132: 		StrictlyProved:  !tied,
133: 		MinimumGapSlack: math.Inf(1),
134: 	}
135: 	if tied {
136: 		certificate.StrictlyProved = false
137: 		certificate.MinimumGapSlack = 0
138: 		// MinimumCap is not applicable to a zero-gap plaintext tie. Keep the
139: 		// numeric evidence JSON-safe and use PlainTie as the fail-closed state.
140: 		certificate.MinimumCap = 0
141: 		return certificate, nil
142: 	}
143: 	for classIndex := range plainLogits {
144: 		if classIndex == top {
145: 			continue
146: 		}
147: 		classGap := plainLogits[top] - plainLogits[classIndex]
148: 		boundSum := errorBounds[top] + errorBounds[classIndex]
149: 		slack := classGap - boundSum
150: 		certificate.MinimumGapSlack = math.Min(
151: 			certificate.MinimumGapSlack,
152: 			slack,
153: 		)
154: 		certificate.MinimumCap = math.Max(
155: 			certificate.MinimumCap,
156: 			boundSum/classGap,
157: 		)
158: 		if !(classGap > boundSum) {
159: 			certificate.StrictlyProved = false
160: 		}
161: 	}
162: 	return certificate, nil
163: }
